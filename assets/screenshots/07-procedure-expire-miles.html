-- ========================================
-- Procedure: Expire Miles (PR_EXPIRE_MILES)
-- Batch job to expire inactive miles
-- Oracle-specific: %TYPE, INDEX BY, cursor FOR loop, DBMS_OUTPUT
-- ========================================

CREATE OR REPLACE PROCEDURE pr_expire_miles(
  p_run_date      IN  DATE DEFAULT SYSDATE,
  p_batch_id      OUT NUMBER,
  p_expired_count OUT NUMBER,
  p_total_miles   OUT NUMBER
) IS
  TYPE t_expiry_rec IS RECORD (
    expiry_id       miles_expiry.expiry_id%TYPE,
    member_id       miles_expiry.member_id%TYPE,
    miles_to_expire NUMBER
  );
  TYPE t_expiry_tab IS TABLE OF t_expiry_rec INDEX BY PLS_INTEGER;
  v_data t_expiry_tab;
BEGIN
  p_batch_id := pkg_batch_processing.start_batch(
    'MILES_EXPIRY', 'Scheduled Miles Expiry');

  -- Collect expiring miles records via cursor FOR loop
  FOR rec IN (
    SELECT expiry_id, member_id,
           (miles_amount - expired_miles) AS miles_to_expire
    FROM miles_expiry
    WHERE status = 'ACTIVE'
      AND expiry_date <= p_run_date
      AND miles_amount > expired_miles
  ) LOOP
    -- Mark as expired
    UPDATE miles_expiry SET
      status = 'EXPIRED', batch_id = p_batch_id,
      processed_date = SYSDATE
    WHERE expiry_id = rec.expiry_id;

    -- Deduct from member balance
    UPDATE members SET
      available_miles = GREATEST(available_miles - rec.miles_to_expire, 0)
    WHERE member_id = rec.member_id;

    -- Notify member of expiration
    pkg_notification.send_notification(
      rec.member_id, 'MILES_EXPIRY',
      'Miles Expiration Notice',
      TO_CHAR(rec.miles_to_expire, '999,999') ||
      ' miles have expired from your account.'
    );

    p_expired_count := p_expired_count + 1;
    p_total_miles   := p_total_miles + rec.miles_to_expire;

    -- Commit every 1000 records
    IF MOD(p_expired_count, 1000) = 0 THEN COMMIT; END IF;
  END LOOP;

  pkg_batch_processing.complete_batch(p_batch_id, p_expired_count, p_expired_count, 0);
  COMMIT;

  DBMS_OUTPUT.PUT_LINE('Expired ' || p_expired_count || ' records, ' || p_total_miles || ' miles');
EXCEPTION WHEN OTHERS THEN ROLLBACK; RAISE;
END pr_expire_miles;
