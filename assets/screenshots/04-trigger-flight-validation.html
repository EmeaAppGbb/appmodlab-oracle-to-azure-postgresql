-- ========================================
-- Trigger: Flight Validation (TRG_FLIGHT_VALIDATION)
-- BEFORE INSERT/UPDATE — enforces business rules
-- Oracle-specific: :NEW/:OLD, RAISE_APPLICATION_ERROR, INSERTING
-- ========================================

CREATE OR REPLACE TRIGGER trg_flight_validation
  BEFORE INSERT OR UPDATE ON flights
  FOR EACH ROW
DECLARE
  v_member_status VARCHAR2(20);
  v_dup_count     NUMBER;
BEGIN
  -- 1. Member must be ACTIVE
  SELECT status INTO v_member_status
  FROM members WHERE member_id = :NEW.member_id;

  IF v_member_status != 'ACTIVE' THEN
    RAISE_APPLICATION_ERROR(-20200,
      'Cannot record flight for inactive member');
  END IF;

  -- 2. Flight date cannot be in the future
  IF :NEW.flight_date > SYSDATE + 1 THEN
    RAISE_APPLICATION_ERROR(-20202,
      'Flight date cannot be in the future');
  END IF;

  -- 3. Not older than 24 months
  IF :NEW.flight_date < ADD_MONTHS(SYSDATE, -24) THEN
    RAISE_APPLICATION_ERROR(-20203,
      'Flight too old to process');
  END IF;

  -- 4. Departure ≠ Arrival
  IF :NEW.departure_airport = :NEW.arrival_airport THEN
    RAISE_APPLICATION_ERROR(-20204,
      'Departure and arrival must differ');
  END IF;

  -- 5. Duplicate check on INSERT
  IF INSERTING THEN
    SELECT COUNT(*) INTO v_dup_count
    FROM flights
    WHERE member_id = :NEW.member_id
      AND flight_number = :NEW.flight_number
      AND flight_date = :NEW.flight_date;
    IF v_dup_count > 0 THEN
      RAISE_APPLICATION_ERROR(-20206, 'Duplicate flight');
    END IF;
  END IF;

  -- 6. Normalize data to uppercase
  :NEW.flight_number     := UPPER(:NEW.flight_number);
  :NEW.airline_code      := UPPER(:NEW.airline_code);
  :NEW.departure_airport := UPPER(:NEW.departure_airport);
  :NEW.arrival_airport   := UPPER(:NEW.arrival_airport);

  -- 7. Enforce minimum 500 miles
  IF :NEW.total_miles < 500 THEN
    :NEW.total_miles := 500;
    :NEW.base_miles  := 500;
  END IF;

END trg_flight_validation;
