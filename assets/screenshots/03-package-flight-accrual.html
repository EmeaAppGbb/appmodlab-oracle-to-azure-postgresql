-- ========================================
-- Package: Flight Accrual (PKG_FLIGHT_ACCRUAL)
-- Processes flight miles — base calc, bonus, bulk accrual
-- Key Oracle features: BULK COLLECT, FORALL, %TYPE, sequences
-- ========================================

CREATE OR REPLACE PACKAGE BODY pkg_flight_accrual AS

  -- Cabin multipliers: Economy 1x, Premium 1.5x, Business 2x, First 3x
  FUNCTION calculate_base_miles(
    p_distance  IN NUMBER,
    p_booking   IN VARCHAR2,
    p_cabin     IN VARCHAR2
  ) RETURN NUMBER IS
  BEGIN
    v_multiplier := CASE p_cabin
      WHEN 'ECONOMY'         THEN 1.0
      WHEN 'PREMIUM_ECONOMY' THEN 1.5
      WHEN 'BUSINESS'        THEN 2.0
      WHEN 'FIRST'           THEN 3.0
    END;

    v_class_factor := CASE
      WHEN p_booking IN ('Y','J','F','C') THEN 1.5   -- Full fare
      WHEN p_booking IN ('B','M','H')     THEN 1.0   -- Standard
      WHEN p_booking IN ('Q','V','W')     THEN 0.75  -- Discount
      WHEN p_booking IN ('L','K','N')     THEN 0.5   -- Deep discount
    END;

    RETURN GREATEST(ROUND(p_distance * v_multiplier * v_class_factor), 500);
  END;

  -- Bulk process pending flights — uses BULK COLLECT + FORALL
  PROCEDURE process_pending_accruals(
    p_batch_size      IN  NUMBER DEFAULT 1000,
    p_processed_count OUT NUMBER
  ) IS
    TYPE t_flight_ids IS TABLE OF flights.flight_id%TYPE;
    v_flight_ids t_flight_ids;
    v_member_ids t_flight_ids;
    v_total_miles t_flight_ids;
  BEGIN
    -- Oracle BULK COLLECT — fetch pending flights
    SELECT flight_id, member_id, total_miles
    BULK COLLECT INTO v_flight_ids, v_member_ids, v_total_miles
    FROM flights
    WHERE accrual_status = 'PENDING'
    FETCH FIRST p_batch_size ROWS ONLY;

    -- Oracle FORALL — bulk update in one round-trip
    FORALL i IN v_flight_ids.FIRST .. v_flight_ids.LAST
      UPDATE flights
      SET accrual_status = 'PROCESSED', processed_date = SYSDATE
      WHERE flight_id = v_flight_ids(i);

    -- Update member balances
    FOR i IN v_member_ids.FIRST .. v_member_ids.LAST LOOP
      pkg_member_mgmt.update_miles_balance(v_member_ids(i), v_total_miles(i), 'EARN');
    END LOOP;

    p_processed_count := v_flight_ids.COUNT;
    COMMIT;
  END;

END pkg_flight_accrual;
